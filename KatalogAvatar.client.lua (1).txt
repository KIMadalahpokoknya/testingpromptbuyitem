local function create(class, props, parent)
    local instance = Instance.new(class)
    for key, value in pairs(props) do
        instance[key] = value
    end
    instance.Parent = parent
    return instance
end

local function corner(parent, radius)
    return create("UICorner", {CornerRadius = UDim.new(0, radius)}, parent)
end

local function stroke(parent, color, transparency)
    return create("UIStroke", {
        Color = color,
        Transparency = transparency,
        Thickness = 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function gradient(parent, a, b)
    return create("UIGradient", {Color = ColorSequence.new(a, b), Rotation = 90}, parent)
end

-- Memisahkan UIGradient tombol dari TextLabel agar warna teks tetap putih bersih
-- dan warna gradien tidak terkalikan ganda dengan BackgroundColor3.
local function createGradientButton(props, textProps, colorA, colorB, radius, parent)
    local buttonProps = {
        BackgroundColor3 = pureWhite,
        BorderSizePixel = 0,
        Text = "",
        AutoButtonColor = true,
    }
    for key, value in pairs(props) do
        buttonProps[key] = value
    end
    local button = create("TextButton", buttonProps, parent)
    corner(button, radius)
    local btnGradient = gradient(button, colorA, colorB)

    local labelProps = {
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        TextColor3 = white,
        Font = Enum.Font.GothamBold,
        TextSize = 15,
    }
    for key, value in pairs(textProps) do
        labelProps[key] = value
    end
    local label = create("TextLabel", labelProps, button)
    return button, label, btnGradient
end

-- Scaling responsif dengan batas minimum agar tidak negatif/terpotong di layar kecil
local cameraConnection
local function updateScale()
    local camera = Workspace.CurrentCamera
    if not camera then return end
    local view = camera.ViewportSize
    if view.X <= 0 or view.Y <= 0 then return end
    local rawScale = math.min(1, (view.X - 24) / 500, (view.Y - 48) / 460)
    panelScale.Scale = math.clamp(rawScale, 0.25, 1)
end

local function bindCamera()
    if cameraConnection then
        cameraConnection:Disconnect()
        cameraConnection = nil
    end
    local camera = Workspace.CurrentCamera
    if camera then
        cameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale)
    end
    updateScale()
end
local workspaceCameraConnection = Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(bindCamera)
bindCamera()

-- Mengekstrak ID dari angka langsung, rbxassetid://, maupun URL katalog/bundle Roblox
local function parseCatalogInput(rawText)
    local trimmed = rawText:match("^%s*(.-)%s*$") or ""
    if trimmed == "" then
        return nil, nil, nil
    end

    local detectedMode = nil
    local rawDigits = trimmed:match("^%d+$")

    if not rawDigits then
        local bundleDigits = trimmed:match("/bundles/(%d+)")
        local catalogDigits = trimmed:match("/catalog/(%d+)")
        local assetUriDigits = trimmed:match("rbxassetid://(%d+)")
            or trimmed:match("[?&]id=(%d+)")

        if bundleDigits then
            rawDigits = bundleDigits
            detectedMode = "Bundle"
        elseif catalogDigits then
            rawDigits = catalogDigits
            detectedMode = "Asset"
        elseif assetUriDigits then
            rawDigits = assetUriDigits
            detectedMode = "Asset"
        else
            return nil, nil, nil
        end
    end

    local normalized = rawDigits:gsub("^0+", "")
    if normalized == "" then
        return nil, nil, nil
    end

    -- Cegah pembulatan diam-diam atau overflow di luar batas integer presisi penuh Luau (2^53 - 1)
    if #normalized > 16 or (#normalized == 16 and normalized > MAX_SAFE_ID_STR) then
        return nil, nil, nil
    end

    local numericId = tonumber(normalized)
    if not numericId or numericId <= 0 then
        return nil, nil, nil
    end

    return numericId, normalized, detectedMode
end

local function fetchAssetInfo(assetId)
    if MarketplaceService.GetProductInfoAsync then
        return MarketplaceService:GetProductInfoAsync(assetId, Enum.InfoType.Asset)
    end
    return MarketplaceService:GetProductInfo(assetId, Enum.InfoType.Asset)
end

local function fetchBundleInfo(bundleId)
    -- Coba AvatarEditorService lebih dulu karena menyertakan harga & status jual bundle
    local okDetails, details = pcall(function()
        if AvatarEditorService.GetItemDetailsAsync then
            return AvatarEditorService:GetItemDetailsAsync(bundleId, Enum.AvatarItemType.Bundle)
        end
        return AvatarEditorService:GetItemDetails(bundleId, Enum.AvatarItemType.Bundle)
    end)
    if okDetails and type(details) == "table" and details.Name then
        return details, "AvatarEditor"
    end

    local okBundle, bundleData = pcall(function()
        return AssetService:GetBundleDetailsAsync(bundleId)
    end)
    if okBundle and type(bundleData) == "table" and bundleData.Name then
        return bundleData, "AssetService"
    end

    return nil, nil
end

local function getPurchaseRemote()
    local remote = ReplicatedStorage:FindFirstChild("KatalogAvatarPurchase")
    if remote and remote:IsA("RemoteEvent") then
        return remote
    end
    return nil
end

local function findItem()
    local id, idStr, detectedMode = parseCatalogInput(input.Text)
    if detectedMode and detectedMode ~= searchMode then
        setSearchMode(detectedMode, false)
    end

    if not id or not idStr then
        requestNumber += 1
        selectedItem = nil
        inFlightQuery = nil
        searchLabel.Text = "Cari"
        image.Image = ""
        placeholder.Visible = true
        itemName.Text = "ID tidak valid"
        unbuyableReason = "Masukkan angka ID atau link katalog yang benar."
        status.Text = unbuyableReason
        setBuyState(false, searchMode == "Bundle" and "Beli Bundle" or "Beli Item")
        return
    end

    local queryKey = searchMode .. ":" .. idStr
    -- Cegah spam request identik saat sedang loading atau diklik beruntun sangat cepat
    if inFlightQuery == queryKey then
        return
    end
    if lastCompletedQuery == queryKey and (os.clock() - lastCompletedAt) < 0.75 then
        return
    end

    requestNumber += 1
    local currentRequest = requestNumber
    inFlightQuery = queryKey
    selectedItem = nil
    unbuyableReason = nil
    image.Image = ""
    placeholder.Visible = true
    searchLabel.Text = "..."
    itemName.Text = "Mencari " .. string.lower(searchMode) .. "..."
    status.Text = "Mohon tunggu sebentar."
    setBuyState(false, searchMode == "Bundle" and "Beli Bundle" or "Beli Item")

    if searchMode == "Bundle" then
        local info, source = fetchBundleInfo(id)
        if requestNumber ~= currentRequest then return end
        inFlightQuery = nil
        searchLabel.Text = "Cari"

        if not info then
            itemName.Text = "Bundle tidak ditemukan"
            unbuyableReason = "Periksa Bundle ID dan coba lagi."
            status.Text = unbuyableReason
            return
        end

        lastCompletedQuery = queryKey
        lastCompletedAt = os.clock()
        itemName.Text = info.Name or "Bundle avatar"
        image.Image = "rbxthumb://type=BundleThumbnail&id=" .. idStr .. "&w=150&h=150"

        if source == "AvatarEditor" then
            if info.Owned == true then
                unbuyableReason = "Kamu sudah memiliki bundle ini."
                status.Text = unbuyableReason
                setBuyState(false, "Sudah Dimiliki")
                return
            end
            local bundleResale = info.LowestResalePrice or info.LowestPrice
            local bundleOffSale = (info.PriceStatus == "Off Sale")
                or (info.PriceStatus == "No Resellers")
                or (info.IsPurchasable == false and bundleResale == nil)
            if bundleOffSale then
                unbuyableReason = "Bundle ini sedang tidak dijual (Off-Sale)."
                status.Text = unbuyableReason
                setBuyState(false, "Tidak Dijual")
                return
            end

            selectedItem = {id = id, kind = "Bundle", isLimited = false}
            setBuyState(true, "Beli Bundle")
            if info.PriceStatus == "Free" or (info.IsPurchasable == true and info.Price == 0) then
                status.Text = "Harga: Gratis (0 Robux)\nHarga final ada di prompt Roblox."
            elseif info.Price ~= nil and info.Price > 0 then
                status.Text = "Harga: " .. tostring(info.Price) .. " Robux\nHarga final ada di prompt Roblox."
            elseif bundleResale ~= nil then
                status.Text = "Harga mulai: " .. tostring(bundleResale) .. " Robux\nHarga final ada di prompt Roblox."
            else
                status.Text = "Cek ketersediaan & harga di prompt Roblox."
            end
        else
            selectedItem = {id = id, kind = "Bundle", isLimited = false}
            setBuyState(true, "Beli Bundle")
            status.Text = "Bundle ditemukan. Cek harga di prompt Roblox."
        end
        return
    end

    -- Mode Pencarian Asset
    local ok, info = pcall(function()
        return fetchAssetInfo(id)
    end)
    if requestNumber ~= currentRequest then return end
    inFlightQuery = nil
    searchLabel.Text = "Cari"

    if not ok or type(info) ~= "table" then
        itemName.Text = "Item tidak ditemukan"
        unbuyableReason = "Periksa Asset ID dan coba lagi."
        status.Text = unbuyableReason
        return
    end

    lastCompletedQuery = queryKey
    lastCompletedAt = os.clock()
    itemName.Text = info.Name or "Item avatar"
    image.Image = "rbxthumb://type=Asset&id=" .. idStr .. "&w=150&h=150"

    local nonAvatarLabel = NON_AVATAR_ASSET_TYPES[info.AssetTypeId]
    if nonAvatarLabel then
        unbuyableReason = "ID ini bertipe " .. nonAvatarLabel .. ", bukan item avatar."
        status.Text = unbuyableReason
        setBuyState(false, "Bukan Item Avatar")
        return
    end

    if info.CanBeSoldInThisGame == false then
        unbuyableReason = "Item ini tidak diizinkan kreatornya dijual di dalam game."
        status.Text = unbuyableReason
        setBuyState(false, "Tidak Dijual di Game")
        return
    end

    local collectibles = type(info.CollectiblesItemDetails) == "table" and info.CollectiblesItemDetails or nil
    local isCollectibleLimited = (collectibles ~= nil and collectibles.IsLimited == true)
        or info.IsLimited == true
        or info.IsLimitedUnique == true

    local lowestResale = collectibles and collectibles.CollectibleLowestResalePrice or nil
    local canPurchase
    if collectibles ~= nil then
        canPurchase = (collectibles.IsForSale == true)
            or (lowestResale ~= nil)
            or (info.IsPublicDomain == true)
    else
        canPurchase = (info.IsForSale == true) or (info.IsPublicDomain == true)
    end

    if not canPurchase then
        unbuyableReason = "Item ini sedang tidak dijual (Off-Sale)."
        status.Text = unbuyableReason
        setBuyState(false, "Tidak Dijual")
        return
    end

    if isCollectibleLimited and not getPurchaseRemote() then
        -- Kebijakan Roblox mewajibkan PromptPurchase UGC Limited dipanggil dari Server
        if collectibles ~= nil and collectibles.IsLimited == true then
            unbuyableReason = "Item UGC Limited butuh KatalogAvatar.server.lua di ServerScriptService."
            status.Text = unbuyableReason
            setBuyState(false, "Perlu Script Server")
            return
        end
    end

    selectedItem = {
        id = id,
        kind = "Asset",
        isLimited = isCollectibleLimited,
    }
    setBuyState(true, "Beli Item")

    local primaryOnSale = (info.IsForSale == true)
        and (collectibles == nil or collectibles.IsForSale == true)

    if info.IsPublicDomain == true or (primaryOnSale and info.PriceInRobux == 0) then
        status.Text = "Harga: Gratis (0 Robux)\nHarga final ada di prompt Roblox."
    elseif primaryOnSale and info.PriceInRobux ~= nil and info.PriceInRobux > 0 then
        status.Text = "Harga: " .. tostring(info.PriceInRobux) .. " Robux\nHarga final ada di prompt Roblox."
    elseif lowestResale ~= nil then
        status.Text = "Harga Resale: " .. tostring(lowestResale) .. " Robux\nHarga final ada di prompt Roblox."
    else
        status.Text = "Harga cek langsung di prompt Roblox."
    end
end

search.Activated:Connect(findItem)
input.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        findItem()
    end
end)

local lastBuyPromptAt = 0
buy.Activated:Connect(function()
    if not selectedItem then
        status.Text = unbuyableReason or "Cari item terlebih dahulu."
        return
    end

    if (os.clock() - lastBuyPromptAt) < 0.75 then
        return
    end
    lastBuyPromptAt = os.clock()

    local current = selectedItem
    local remote = getPurchaseRemote()

    local ok, err = pcall(function()
        if current.kind == "Bundle" then
            if remote then
                remote:FireServer("Bundle", current.id)
            else
                MarketplaceService:PromptBundlePurchase(player, current.id)
            end
        else
            if remote and current.isLimited then
                remote:FireServer("Asset", current.id)
            else
                MarketplaceService:PromptPurchase(player, current.id)
            end
        end
    end)

    if not ok then
        status.Text = "Gagal membuka prompt pembelian."
        warn("Prompt pembelian gagal:", err)
    end
end)

local assetFinishedConn = MarketplaceService.PromptPurchaseFinished:Connect(function(plr, assetId, wasPurchased)
    if plr == player and selectedItem and selectedItem.kind == "Asset" and selectedItem.id == assetId and wasPurchased then
        status.Text = "Pembelian berhasil! Item telah masuk ke inventori kamu."
    end
end)

local bundleFinishedConn = MarketplaceService.PromptBundlePurchaseFinished:Connect(function(plr, bundleId, wasPurchased)
    if plr == player and selectedItem and selectedItem.kind == "Bundle" and selectedItem.id == bundleId and wasPurchased then
        status.Text = "Pembelian bundle berhasil! Cek inventori avatar kamu."
    end
end)

gui.Destroying:Connect(function()
    if cameraConnection then
        cameraConnection:Disconnect()
        cameraConnection = nil
    end
    workspaceCameraConnection:Disconnect()
    assetFinishedConn:Disconnect()
    bundleFinishedConn:Disconnect()
end)
